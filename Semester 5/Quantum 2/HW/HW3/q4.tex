\subsection{0-th Order Hamiltonian}
Given that $\Omega \gg \Delta$, The 0th order hamiltonian is just the element that doesn't include $\Delta$
\begin{equation}
	H = \frac{1}{2}\hbar\Omega\sigma_x
\end{equation}
and to find the time development of the states, since the hamiltonian is time indepdendent
\begin{align}
	U(t) &= e^{-\frac{i}{\hbar}Ht}=e^{-\frac{1}{2}i\Omega\sigma_x}\\
	&= \cos\ins{\frac{\Omega t}{2}}\mathds{1} -i\sin\ins{\frac{\Omega t}{2}}\sigma_x\\
	&= \cos\ins{\frac{\Omega t}{2}}\ins{\ketbra{g}{g}+\ketbra{e}{e}}-i\sin\ins{\frac{\Omega t}{2}}\ins{\ketbra{g}{e}+\ketbra{e}{g}}
\end{align}
Therefore 
\begin{align}
	\ket{g(t)}&=U(t)\ket{g}\\
	&=\cos\ins{\frac{\Omega t}{2}}\ket{g}-i\sin\ins{\frac{\Omega t}{2}}\ket{e}\\
	\ket{e(t)}&=U(t)\ket{e}\\
	&=\cos\ins{\frac{\Omega t}{2}}\ket{e}-i\sin\ins{\frac{\Omega t}{2}}\ket{g}
\end{align}

\subsection{After Half-Pi Pulse}
The states will be
\begin{align}
	\ket{g'} &= \cos\ins{\frac{\pi}{4}}\ket{g}-i\sin\ins{\frac{\pi}{4}}\ket{e}\\
	&= \frac{1}{\sqrt{2}}\ket{g}-\frac{i}{\sqrt{2}}\ket{e}\\
	\ket{e'} &= \cos\ins{\frac{\pi}{4}}\ket{e}-i\sin\ins{\frac{\pi}{4}}\ket{g}\\
	&= \frac{1}{\sqrt{2}}\ket{e}-\frac{i}{\sqrt{2}}\ket{g}
\end{align}
\subsection{Free Development}
When the electric field is turned off, the hamiltonian is
\begin{equation}
	H=\frac{1}{2}\hbar\Delta\sigma_z
\end{equation}
so the time development operator is
\begin{align}
	U_2(t) &= e^{-\frac{i}{\hbar}Ht}=e^{-\frac{1}{2}i\Delta\sigma_z}\\
	&= \cos\ins{\frac{\Delta t}{2}}\mathds{1}+i\sin\ins{\frac{\Delta t}{2}}\sigma_z
\end{align}
therefore
\begin{align}
	\ket{g''} &= U_2(t)\ket{g'}\\
	&= \cos\ins{\frac{\Delta T}{2}}\ket{g'}+i\sin\ins{\frac{\Delta T}{2}}\ins{\ketbra{e}{e}-\ketbra{g}{g}}\ket{e'}\\
	&= \cos\ins{\frac{\Delta T}{2}}\ket{g'}+i\sin\ins{\frac{\Delta T}{2}}\ins{\ketbra{e}{e}-\ketbra{g}{g}}\ins{\frac{1}{\sqrt{2}}\ket{e}-\frac{i}{\sqrt{2}}\ket{g}}\\
	&= \cos\ins{\frac{\Delta T}{2}}\ket{g'}+i\sin\ins{\frac{\Delta T}{2}}\ins{\frac{1}{\sqrt{2}}\ket{e}+\frac{i}{\sqrt{2}}\ket{g}}\\
	&= \frac{1}{\sqrt{2}}\ins{\cos\ins{\frac{\Delta T}{2}}\ins{\ket{g}-\ket{e}}+i\sin\ins{\frac{\Delta T}{2}}\ins{\ket{e}+\ket{g}}}\\
	&= \frac{1}{\sqrt{2}}\ins{\ins{\cos\ins{\frac{\Delta T}{2}}+i\sin\ins{\frac{\Delta T}{2}}}\ket{g}-\ins{\cos\ins{\frac{\Delta T}{2}}-i\sin\ins{\frac{\Delta T}{2}}}\ket{e}}\\
	&= \frac{1}{\sqrt{2}}\ins{e^{\frac{i\Delta t}{2}}\ket{g}-ie^{-\frac{i\Delta t}{2}}\ket{e}}
\end{align}

\subsection{Another Pulse}
We are back to using the electric field time propagator, therefore
\begin{align}
	\ket{g'''} &= U(\frac{\pi}{2\Omega})\ket{g''}\\
	&= \ins{\cos\ins{\frac{\pi}{4}}\ins{\ketbra{g}{g}+\ketbra{e}{e}}-i\sin\ins{\frac{\pi}{4}}\ins{\ketbra{g}{e}+\ketbra{e}{g}}}\frac{1}{\sqrt{2}}\ins{e^{\frac{i\Delta t}{2}}\ket{g}-ie^{-\frac{i\Delta t}{2}}\ket{e}}\\
	&= \frac{1}{\sqrt{2}}\ins{\cos\ins{\frac{\pi}{4}}\ins{e^{\frac{i\Delta t}{2}}\ket{g}-ie^{-\frac{i\Delta t}{2}}\ket{e}}-i\sin\ins{\frac{\pi}{4}}\ins{e^{\frac{i\Delta t}{2}}\ket{e}-ie^{-\frac{i\Delta t}{2}}\ket{g}}}\\
	&= \frac{1}{2}\ins{\ins{e^{\frac{i\Delta t}{2}}-e^{-\frac{i\Delta t}{2}}}\ket{g}-i\ins{e^{\frac{i\Delta t}{2}}+e^{-\frac{i\Delta t}{2}}}\ket{e}}\\
	&= \sinh\ins{\frac{i\Delta t}{2}}\ket{g}-i\cosh\ins{\frac{i\Delta t}{2}}\ket{e}\\
	&= i\sin\ins{\frac{\Delta t}{2}}\ket{g}-i\cos\ins{\frac{\Delta t}{2}}\ket{e}
\end{align}
therefore
\begin{align}
	P_e(\Delta) &= \lvert\braket{e|g'''}\rvert^2\\
	&= \cos^2\ins{\frac{\Delta t}{2}}
\end{align}

\subsection{Tuning and Sensitivity}
\subsubsection{Meaning}

The meaning of the sensitivity S is how strongly a deviation from the resonant frequency affects the operation of the clock. A high value means even small deviations will cause significant changes to the probability of the states flipping.

\subsubsection{Our Clock's Sensitivity}

\begin{equation}
	S = \left\lvert\diff{P_e(\Delta)}{\Delta}\right\rvert=\left\lvert=t\sin\ins{\frac{\Delta t}{2}}\cos\ins{\frac{\Delta t}{2}}\right\rvert = \left\lvert \frac{t}{2}\sin\ins{\Delta t}\right\rvert 
\end{equation}

\subsubsection{Maximum Sensitivity}

We can see the maximums will be found at $t=\frac{(2k+1)\pi}{2\Delta}$

\subsubsection{Error Range}

\begin{align}
	\delta\Delta &= \frac{3\pi-\pi}{2T}=\frac{\pi}{T}
\end{align}

\subsection{Time Until 1s Error}
Using the given equation
\begin{equation}
	\delta t = \frac{\delta f}{f_0}t \implies t=\frac{f_0\delta t}{\delta f}
\end{equation}
therefore
\begin{align}
	t &= \frac{\SI{9192631770}{\hertz}\cdot \SI{1}{\second}}{\SI{\pi}{\per\second}}= \SI{92.72}{\year}
\end{align}
