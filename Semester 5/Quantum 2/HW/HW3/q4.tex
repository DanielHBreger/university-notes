\section{Question 4}
\begin{colbox}
	\begin{enumerate}
		\item Show that the set of states $A$ is an eigenbasis of $H_0 = H_A+H_F$. Find the eigenvalues.
		\item Draw qualitatively on the energy axis the location of the five first eigenenergies of $H_0$. Show that aside from the ground state, $H_0$'s eigenstates are bunched in pairs of degenerate states.
		\item The coupling between the field and the atom is a dipole coupling and is given by
		\begin{equation*}
			W = \gamma\ins{a\sigma_++a^\dagger\sigma_-}
		\end{equation*}
		where $\gamma = -d\sin\ins{kz_0}\sqrt{\hbar\omega/\epsilon_0V}$ where d is the dipole moment, determined experimentally.
		\begin{enumerate}
			\item Write the action of $\hat W$ on the states $\ket{g,n}$ and $\ket{e,n}$
			\item Which physical processes to the operators $a\sigma_+$ and $a^\dagger\sigma_-$ describe?
		\end{enumerate}
	\end{enumerate}
\end{colbox}
\subsection{Eigenbasis}
I'll assume the meaning of the question is to prove the states in set A are eigenstates of the hamiltonian, not to prove they span the energy space.
\begin{align}
	H_0\ket{g,n} &= \ins{\frac{\hbar\omega_A}{2}\sigma_z+\frac{\hbar\omega}{2}\ins{\chi^2+\Pi^2}}\ket{g,n}\\
	&= -\frac{\hbar\omega_A}{2}+\hbar\omega\ins{n+\frac{1}{2}}\ket{g,n}
\end{align}
and
\begin{align}
	H_0\ket{e,n} &= \ins{\frac{\hbar\omega_A}{2}\sigma_z+\frac{\hbar\omega}{2}\ins{\chi^2+\Pi^2}}\ket{e,n}\\
	&= \frac{\hbar\omega_A}{2}+\hbar\omega\ins{n+\frac{1}{2}}\ket{e,n}
\end{align}

\subsection{Drawing}
Since $\omega_A=\omega$ the energies are now
\begin{align}
	H_0\ket{g,n} = \hbar\omega n\ket{g,n} && H_0\ket{e,n} = \hbar\omega\ins{n+1}\ket{e,n}
\end{align}
we can see that the ground state energy is $0$, then every energy level has is degenerate with 2 options for achieving that energy.

\subsection{Coupling}
\subsubsection{Action of W}
\begin{align}
	W\ket{g,n} &= \gamma\ins{a\sigma_++a^\dagger\sigma_- }\ket{g,n}\\
	&= \gamma a\sigma_+\ket{g,n}\\
	&= \gamma \sqrt{n} \ket{e,n - 1}\\
	W\ket{e,n} &= \gamma\ins{a\sigma_++a^\dagger\sigma_- }\ket{e,n}\\
	&= \gamma a^\dagger\sigma_-\ket{g,n}\\
	&= \gamma\sqrt{n+1}\ket{g,n+1}
\end{align}
and of course the $W\ket{g,0}=0$.
\subsubsection{Physical Process}
$a\sigma_+$ desribes a photon being absorbed by the atom raising it to the excited state. $a^\dagger\sigma_-$ describes the opposite.
