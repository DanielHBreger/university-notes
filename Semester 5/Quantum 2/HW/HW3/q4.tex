		\begin{equation*}
			W = \gamma\ins{a\sigma_++a^\dagger\sigma_-}
		\end{equation*}
		where $\gamma = -d\sin\ins{kz_0}\sqrt{\hbar\omega/\epsilon_0V}$ where d is the dipole moment, determined experimentally.
		\begin{enumerate}
			\item Write the action of $\hat W$ on the states $\ket{g,n}$ and $\ket{e,n}$
			\item Which physical processes to the operators $a\sigma_+$ and $a^\dagger\sigma_-$ describe?
		\end{enumerate}
	\end{enumerate}
