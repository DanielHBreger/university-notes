\section{Delta function identities}
\begin{colbox}
	\begin{enumerate}
		\item show the delta function is symmetric
		\item Show \begin{equation}
			\delta\ins{\alpha x} = \frac{\delta\ins{x}}{\abs{\alpha}};\quad \alpha\neq0
		\end{equation}
		\item show \begin{equation}
			\delta\ins{g(x)} = \sum_{i=1}^{N}\frac{\delta\ins{x-x_i}}{\abs{g'(x_i)}}
		\end{equation}
		\item find \begin{equation}
			\int\dd x f(x)\delta'(x)
		\end{equation}
		\item show that \begin{equation}
			\delta(x) = \frac{1}{2\pi}\int_{-\infty}^{\infty}\dd ke^{-ikx}
		\end{equation}
	\end{enumerate}
\end{colbox}
\subsection{symmetry}
Let \(f(x)\) be an arbitrary function. Therefore,

\begin{align*}
    \int\limits_{-\infty}^\infty f(x)\delta(-x)dx &= \left[\begin{aligned}
        u&=-x\\du&=-dx
    \end{aligned}\right]=\int\limits_{\infty}^{-\infty} -f(-u)\delta(u)du\\
    &= \int\limits_{-\infty}^\infty f(-u)\delta(u)du\\
    &= f(-0) = f(0) = \int\limits_{-\infty}^\infty f(x)\delta(x)dx
\end{align*}
Therefore the delta function is symmetric.
\subsection{multiplication}
\begin{align*}
    \int\limits_{-\infty}^\infty f(x)\delta(\alpha x)dx &= \left[\begin{aligned}
        u&=\alpha x\\du&=\alpha dx
    \end{aligned}\right] = \int\limits_{-sgn(\alpha)\cdot\infty}^{sgn(\alpha)\cdot\infty} \frac{1}{\alpha}f\left(\frac{u}{\alpha}\right)\delta(u) du\\
    &= \frac{sgn(\alpha)}{\alpha}f\left(\frac{0}{\alpha}\right)\\
    &= \frac{1}{|\alpha|}f(0)
\end{align*}
Therefore
\begin{equation*}
    \delta(\alpha x)=\frac{\delta(x)}{|\alpha|}
\end{equation*}
\subsection{delta of function}
We'll develop \(g(x)\) into a taylor series around \(x_i\):
\begin{equation*}
    g(x) \approx g(x_i)+g'(x_i)\cdot(x-x_i) = g'(x_i)\cdot(x-x_i)
\end{equation*}
ignoring elements beyond first order. Next we'll look at the following integral:
\begin{equation*}
    \int\limits_{-\infty}^\infty\delta(g(x))f(x)dx
\end{equation*}
where \(f(x)\) is an arbitrary function. It's clear from the definition of \(g(x)\) and the delta function that the integrand is non-zero only at the points \(x_i\), therefore we can integrate only around those points. We'll choose a small enough neighborhood for each point so they are isolated:
\begin{equation*}
    \int\limits_{-\infty}^\infty\delta(g(x))f(x)dx =  \sum\limits_{i=1}^N\int\limits_{x_i-\epsilon_i}^{x_i+\epsilon_i}\delta(g(x))f(x)dx
\end{equation*}
Using the delta function definition and the previous subsection we can find:
\begin{align*}
    \sum\limits_{i=1}^N\int\limits_{x_i-\epsilon_i}^{x_i+\epsilon_i}\delta(g(x))f(x)dx &= \sum\limits_{i=1}^N\int\limits_{x_i-\epsilon_i}^{x_i+\epsilon_i}\delta(g'(x_i)\cdot(x-x_i))f(x)dx\\
    &= \left[\begin{aligned}
        u &= g'(x_i)\cdot(x-x_i)\\
        du &= g'(x_i)dx
    \end{aligned}\right]\\ &= \sum\limits_{i=1}^N\int\limits_{sgn(g'(x_i))\cdot g(x_i-\epsilon_i)}^{sgn(g'(x_i))\cdot g(x_i+\epsilon_i)}\frac{1}{g'(x_i)}\delta(u)f\left(x_i+\frac{u}{g'(x_i)}\right)du\\
    &= \sum\limits_{i=1}^N\frac{1}{|g'(x_i)|}\int\limits_{g(x_i-\epsilon_i)}^{g(x_i+\epsilon_i)}\delta(u)f\left(x_i+\frac{u}{g'(x_i)}\right)du\\
    &= \sum\limits_{i=1}^N\frac{f(x_i)}{|g'(x_i)|}\\
    &= \sum\limits_{i=1}^N\int\limits_{-\infty}^\infty f(x)\frac{\delta(x-x_i)}{|g'(x_i)|}
\end{align*}
Therefore
\begin{equation*}
    \delta(g(x)) = \sum\limits_{i=1}^N\frac{\delta(x-x_i)}{|g'(x_i)|}
\end{equation*}
\subsection{derivative of delta}
Integrating by parts we get
\begin{align*}
    \int\limits_{-\infty}^\infty f(x)\delta'(x)dx &= \cancelto{0}{\left.f(x)\delta(x)\right|_{-\infty}^\infty}-\int\limits_{-\infty}^\infty f'(x)\delta(x)\\
    &= -f'(0)
\end{align*}
\subsection{fourier transform}
If we perform a fourier transform on the delta function we get
\begin{align}
	\mathcal{F}\delta(x) &= \frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}\delta(x)e^{iky}\dd y = \left.\frac{1}{\sqrt{2\pi}}e^{iky}\right\vert_{y=0}=\frac{1}{\sqrt{2\pi}}
\end{align}
therefore if we apply the inverse fourier transform
\begin{align}
	\delta(x) = \frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty}\mathcal{F}\delta(y)e^{iky} \dd y = \frac{1}{2\pi}\int_{-\infty}^{\infty}e^{iky} \dd y
\end{align}
and if we rename the parameters
\begin{equation}
	\delta(x)= \frac{1}{2\pi}\int_{-\infty}^{\infty}e^{ikx} \dd k
\end{equation}
